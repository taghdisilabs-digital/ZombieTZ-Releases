def digest(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def validate_plan(value):
    if not isinstance(value, dict) or set(value) != {"argv", "outputs", "timeout_seconds"}:
        raise ValueError("plan requires argv, outputs and timeout_seconds")
    argv = value["argv"]
    if (not isinstance(argv, list) or not argv or len(argv) > 256
            or any(not isinstance(x, str) or "\x00" in x or len(x) > 65536 for x in argv)):
        raise ValueError("invalid direct executable argv")
    executable = Path(argv[0])
    if not executable.is_absolute() or not executable.resolve().is_relative_to("/usr"):
        raise ValueError("executable must be an installed /usr tool")
    if not executable.is_file():
        raise ValueError("executable is unavailable")
    outputs = value["outputs"]
    if not isinstance(outputs, dict) or not 1 <= len(outputs) <= 32:
        raise ValueError("one or more exact output expectations required")
    for name, sha in outputs.items():
        if not isinstance(name, str) or "\x00" in name:
            raise ValueError("invalid output path")
        p = PurePosixPath(name)
        if p.is_absolute() or ".." in p.parts or not p.parts or name != p.as_posix():
            raise ValueError("output path must stay inside the project")
        if not isinstance(sha, str) or len(sha) != 64 or any(c not in "0123456789abcdef" for c in sha):
            raise ValueError("exact output SHA-256 required")
    timeout = value["timeout_seconds"]
    if isinstance(timeout, bool) or not isinstance(timeout, (int, float)) or not math.isfinite(timeout) or not 0 < timeout <= 600:
        raise ValueError("timeout must be positive and at most the 600-second operation bound")
    return json.loads(json.dumps(value))

def _workspace(store, project):
    value = project.get("workspace_path")
    if not value or not Path(value).is_absolute():
        raise ValueError("Project requires an absolute workspace")
    work = Path(value).resolve()
    authority = store.database_path.resolve().parent
    if authority.is_relative_to(work) or work.is_relative_to(authority):
        raise ValueError("Controller state must be outside the executable workspace")
    return work

def _parent_binding(store, project_id, task_id, expected_revision):
    if isinstance(expected_revision, bool) or not isinstance(expected_revision, int):
        raise ValueError("Exact parent revision required")
    access, parent = _native_task(store, task_id)
    projected = store.task(task_id)
    if parent.project_ref.value != project_id:
        raise ValueError("Parent must belong to the same project")
    if parent.revision != expected_revision:
        raise ValueError("Parent revision binding changed")
    if parent.side_effect_authority != "PROJECT_WRITE":
        raise ValueError("Parent authority does not permit project writes")
    if parent.constraints.get("parent_task_id"):
        raise ValueError("Nested execution steps are not supported")
    if projected["status"] in {"SUCCEEDED", "FAILED", "CANCELLED"}:
        raise ValueError("Parent task is already terminal")
    return {"parent_task_id": parent.task_id, "parent_task_revision": parent.revision,
            "parent_task_digest": parent.canonical_digest}

def _verify_parent(store, task):
    parent_id = task.constraints.get("parent_task_id")
    if parent_id:
        actual = _parent_binding(store, task.project_ref.value, parent_id,
                                 task.constraints["parent_task_revision"])
        if any(task.constraints.get(key) != value for key, value in actual.items()):
            raise ValueError("Parent task binding changed")

def create_process_task(store, project_id, intent, plan, *, parent_task_id=None, expected_parent_revision=None):
    plan = validate_plan(plan)
    if parent_task_id is not None and (not isinstance(parent_task_id, str) or not parent_task_id):

    parent = _parent_binding(store, project_id, parent_task_id, expected_parent_revision) if parent_task_id else {}
    if expected_parent_revision is not None and not parent_task_id:
        raise ValueError('Parent revision requires a parent task')
    if not str(intent).strip() or len(intent) > 8000:
        raise ValueError("task intent is invalid")
    project = store.project(project_id)
    workspace = _workspace(store, project)
    workspace.mkdir(parents=True, exist_ok=True)
    access = store._access(project_id)
    objects = FilesystemObjectStorageBackend(store.database_path.parent / "objects")
    process = ManagedProcessAdapter(store.database_path, objects, inherited_environment={})
    capability = process.capability_ref(shell=False)

    packed = json.dumps(plan, sort_keys=True, separators=(",", ":"))

def supports_task(task):

def _native_task(store, task_id):

    access = store._access(projected["project_id"])
    return access, next(t for t in store.tasks.list_tasks(access, access.project_ref)

def run_process_task(store, task_id, runtime_root, *, cancelled=None):
    access, task = _native_task(store, task_id)
    if task.task_type != TASK_TYPE:
        raise ValueError("Task has no admitted native process plan")
    plan = validate_plan(json.loads(task.constraints[PLAN_KEY]))
    project = store.project(access.project_ref.value)
    work = _workspace(store, project)
    if str(work) != task.constraints["workspace"]:
        raise ValueError("Project workspace binding changed")
    if digest(plan["argv"][0]) != task.constraints["executable_sha256"]:
        raise ValueError("Executable changed after plan binding")
    root = Path(runtime_root).resolve() / "native" / task_id
    if root.is_relative_to(work):
        raise ValueError("Controller state must be outside the executable workspace")
    root.mkdir(parents=True, exist_ok=True)
    locks = store.database_path.resolve().parent / "native-dispatch-locks"
    locks.mkdir(exist_ok=True)
    with (locks / (access.project_ref.value + ".lock")).open("a") as lock:
        fcntl.flock(lock, fcntl.LOCK_EX)
        if cancelled is not None and cancelled():
            return {"status": "CANCELLED", "task_id": task_id}
        latest_access, latest = _native_task(store, task_id)
        if latest.task_ref != task.task_ref or latest.canonical_digest != task.canonical_digest:
            raise ValueError("Task revision changed before dispatch")
        prior = [r for r in store.runs.list_runs(access, access.project_ref)
                 if r.task_ref == task.task_ref]
